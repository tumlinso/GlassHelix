cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(GlassHelixSystemQualification LANGUAGES CXX)
 include(CTest)
 get_filename_component(GH_SYSTEM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
else()
 set(GH_SYSTEM_ROOT "${PROJECT_SOURCE_DIR}")
endif()
if(NOT EXISTS "${GH_CONTRACT_ROOT}/include/GlassHelix/core/system.hh" OR NOT EXISTS "${CELLERATOR_SOURCE_ROOT}/include/Cellerator/compute/operation/native_foundation_contract.hh")
 message(FATAL_ERROR "Explicit integrated scientific and Cellerator contract sources required")
endif()
add_library(gh_scientific_assembly STATIC "${GH_SYSTEM_ROOT}/src/system/assembly.cc")
target_compile_features(gh_scientific_assembly PUBLIC cxx_std_20)
target_include_directories(gh_scientific_assembly PUBLIC "${GH_SYSTEM_ROOT}/include" "${GH_CONTRACT_ROOT}/include" "${CELLERATOR_SOURCE_ROOT}/include")
# Qualify against the actual existing CE semantic/program implementation.
add_library(ce_assembly_owners STATIC "${CELLERATOR_SOURCE_ROOT}/src/compute/operation/operation_core_v2/schema.cc" "${CELLERATOR_SOURCE_ROOT}/src/execution/program/program_v2.cc")
target_compile_features(ce_assembly_owners PUBLIC cxx_std_20)
target_include_directories(ce_assembly_owners PUBLIC "${CELLERATOR_SOURCE_ROOT}/include")
add_executable(gh_nf1_s01 s01.cc)
target_link_libraries(gh_nf1_s01 PRIVATE gh_scientific_assembly ce_assembly_owners)
add_test(NAME gh_nf1_s01 COMMAND gh_nf1_s01)
