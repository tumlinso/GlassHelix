cmake_minimum_required(VERSION 3.28)
project(GlassHelixInstalledContracts LANGUAGES CXX)
find_package(GlassHelixSubstrate CONFIG REQUIRED COMPONENTS contracts)
if(NOT GLASSHELIX_CELLERATOR_INCLUDE)
  message(FATAL_ERROR "An actual published CE header directory is required")
endif()
add_executable(installed_contracts consumer.cc)
target_link_libraries(installed_contracts PRIVATE GlassHelixSubstrate::contracts)
target_include_directories(installed_contracts PRIVATE "${GLASSHELIX_CELLERATOR_INCLUDE}")
