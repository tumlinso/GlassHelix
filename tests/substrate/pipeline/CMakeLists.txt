cmake_minimum_required(VERSION 3.28)
project(GHPipelineConsumer LANGUAGES CXX)
find_package(Cellerator CONFIG REQUIRED COMPONENTS adaptive)
if(NOT Cellerator_BUILD_SOURCE_REVISION STREQUAL CE_EXPECTED_REVISION)
  message(FATAL_ERROR "Installed CE source revision differs from the prerequisite")
endif()
add_executable(native_refactor native_refactor.cc)
target_compile_features(native_refactor PRIVATE cxx_std_20)
target_link_libraries(native_refactor PRIVATE Cellerator::adaptive)
