cmake_minimum_required(VERSION 3.20)
project(GlassHelixNF1Example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GH_NF1_REFERENCE_ONLY "Build only the independent formula oracle, NOT GlassHelix" OFF)
enable_testing()
if(GH_NF1_REFERENCE_ONLY)
  add_executable(nf1_reference reference_main.cc)
  add_test(NAME nf1_reference_mathematics COMMAND nf1_reference)
  return()
endif()
if(NOT TARGET GlassHelix::glasshelix)
  find_package(GlassHelix CONFIG REQUIRED)
endif()
add_executable(mechanistic_toolbox mechanistic_toolbox.cc)
target_link_libraries(mechanistic_toolbox PRIVATE GlassHelix::glasshelix)
add_test(NAME gh_nf1_demo_cuda COMMAND mechanistic_toolbox)
add_test(NAME gh_nf1_demo_cpu COMMAND mechanistic_toolbox --cpu)
